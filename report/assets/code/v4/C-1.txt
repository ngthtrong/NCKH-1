 # 1. Cơ sở dữ liệu quan hệ PostgreSQL 16 (0.50 vCPU, 1.536 MB RAM)
 postgres-db:
   image: postgres:16-alpine
   container_name: tenantflow-postgres
   environment:
     POSTGRES_DB: kanban_saas_db
     POSTGRES_USER: postgres
     POSTGRES_PASSWORD: <GIÁ TRỊ CẤU HÌNH RIÊNG>
   volumes:
     - pgdata:/var/lib/postgresql/data
     - ./infra/db/init-roles.sql:/docker-entrypoint-initdb.d/01-init-roles.sql:ro # Khởi tạo role NOBYPASSRLS
   networks:
     - nckh-ec2-net
   healthcheck:
     test: ["CMD-SHELL", "pg_isready -U postgres -d kanban_saas_db"]
     interval: 5s
     retries: 10
   deploy:
     resources:
       limits: { cpus: "0.50", memory: 1536M }
 # 2. Bộ nhớ đệm phân tán & Token Bucket Redis 7.2 (0.25 vCPU, 512 MB RAM)
 redis-cache:
   image: redis:7.2-alpine
   container_name: tenantflow-redis
   command: ["redis-server", "--appendonly", "yes"]
   networks:
     - nckh-ec2-net
   healthcheck:
     test: ["CMD", "redis-cli", "ping"]
     interval: 5s
     retries: 10
   deploy:
     resources:
       limits: { cpus: "0.25", memory: 512M }
 # 3. Dịch vụ ứng dụng backend Spring Boot 3.3 (1.25 vCPU, 2.560 MB RAM)
 kanban-api:
   build: { context: ./apps/api, dockerfile: Dockerfile }
   container_name: tenantflow-api
   environment:
     SPRING_DATASOURCE_URL: jdbc:postgresql://postgres-db:5432/kanban_saas_db
     SPRING_DATASOURCE_USERNAME: kanban_app_user
     SPRING_DATASOURCE_PASSWORD: <GIÁ TRỊ CẤU HÌNH RIÊNG>
     SPRING_DATASOURCE_HIKARI_MAXIMUM_POOL_SIZE: 25
     SPRING_DATA_REDIS_HOST: redis-cache
     SPRING_DATA_REDIS_PORT: 6379
     APP_JWT_SECRET: <GIÁ TRỊ CẤU HÌNH RIÊNG>
   depends_on:
     postgres-db: { condition: service_healthy }
     redis-cache: { condition: service_healthy }
   networks:
     - nckh-ec2-net
   deploy:
     resources:
       limits: { cpus: "1.25", memory: 2560M }
