-- Cấp quyền thao tác cho tài khoản ứng dụng
GRANT SELECT, INSERT, UPDATE, DELETE
    ON tenants, tenant_users, boards, board_columns, cards
    TO kanban_app_user;


-- Kích hoạt và cưỡng chế RLS
ALTER TABLE boards ENABLE ROW LEVEL SECURITY;
ALTER TABLE boards FORCE ROW LEVEL SECURITY;

ALTER TABLE board_columns ENABLE ROW LEVEL SECURITY;
ALTER TABLE board_columns FORCE ROW LEVEL SECURITY;

ALTER TABLE cards ENABLE ROW LEVEL SECURITY;
ALTER TABLE cards FORCE ROW LEVEL SECURITY;


-- Chính sách cô lập dữ liệu bảng boards
CREATE POLICY p_boards_isolation
    ON boards
    FOR ALL
    TO kanban_app_user
    USING (
        tenant_id = NULLIF(
            current_setting('app.current_tenant_id', true), ''
        )::uuid
    )
    WITH CHECK (
        tenant_id = NULLIF(
            current_setting('app.current_tenant_id', true), ''
        )::uuid
    );
