// 1. Quản lý ngữ cảnh luồng cô lập (vn/tenantflow/security/TenantContext.java)
public final class TenantContext {
    private static final ThreadLocal<TenantInfo> CONTEXT = new ThreadLocal<>();
    public static void set(TenantInfo info) { CONTEXT.set(info); }
    public static TenantInfo get() { return CONTEXT.get(); }
    public static void clear() { CONTEXT.remove(); } // BẮT BUỘC xóa sạch để chống ô nhiễm Thread Pool
}

// 2. Trích đoạn kiểm tra biên tại JwtTenantFilter.java
String hostSubdomain = extractSubdomain(request.getHeader("Host")); // VD: "t-acme"
if (hostSubdomain != null && !hostSubdomain.equalsIgnoreCase(info.slug())) {
    response.sendError(HttpServletResponse.SC_FORBIDDEN, "Subdomain mismatch with JWT"); // Chặn 403
    return;
}
try {
    TenantContext.set(info);
    MDC.put("tenant_id", info.tenantId().toString());
    chain.doFilter(request, response);
} finally {
    TenantContext.clear(); // Giải phóng ThreadLocal trước khi trả luồng về Tomcat Pool
    MDC.remove("tenant_id");
}
