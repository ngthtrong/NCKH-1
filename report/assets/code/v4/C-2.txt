{   # Kiểm duyệt tên miền hợp lệ trước khi cấp phát chứng chỉ HTTPS tự động
    admin off
    on_demand_tls { ask http://kanban-api:8080/api/public/tls-check }
}

# 1. Không gian thuê bao công cộng (áp dụng tương tự cho admin.tenantflow.tech)
*.{$BASE_DOMAIN:tenantflow.tech} {
    tls { on_demand }
    encode zstd gzip
    handle /api/* {
        reverse_proxy kanban-api:8080 { header_up Host {http.request.host} }
    }
    handle { reverse_proxy kanban-web:80 }
}

# 2. Cổng HTTP nội bộ riêng cho công cụ đo tải k6 (tránh chuyển hướng 308 sang HTTPS)
:8080 {
    handle /api/* {
        reverse_proxy kanban-api:8080 { header_up Host {http.request.host} }
    }
}
