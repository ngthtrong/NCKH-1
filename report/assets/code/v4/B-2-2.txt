-- KEYS[1] = ratelimit:tenant:{tenant_id} | ARGV = {capacity, refill_per_sec, now_ms, cost}
local key, cap, rate = KEYS[1], tonumber(ARGV[1]), tonumber(ARGV[2])
local now, req       = tonumber(ARGV[3]), tonumber(ARGV[4])
local data   = redis.call('HMGET', key, 'tokens', 'last_refill')
local tokens = tonumber(data[1]) or cap
local last   = tonumber(data[2]) or now
tokens = math.min(cap, tokens + math.max(0, (now - last) / 1000.0) * rate)
if tokens < req then return { 0, math.floor(tokens) } end -- Cạn token -> HTTP 429
tokens = tokens - req
redis.call('HMSET', key, 'tokens', tokens, 'last_refill', now)
redis.call('EXPIRE', key, 120)
return { 1, math.floor(tokens) }
