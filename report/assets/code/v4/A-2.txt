-- Bảng Kanban
CREATE TABLE boards (
    id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id  UUID NOT NULL
               REFERENCES tenants(id) ON DELETE CASCADE,
    name       VARCHAR(150) NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_boards_tenant_created
    ON boards (tenant_id, created_at DESC);


-- Bảng cột trạng thái
CREATE TABLE board_columns (
    id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id  UUID NOT NULL
               REFERENCES tenants(id) ON DELETE CASCADE,
    board_id   UUID NOT NULL
               REFERENCES boards(id) ON DELETE CASCADE,
    name       VARCHAR(80) NOT NULL,
    position   INT NOT NULL,
    wip_limit  INT NULL
               CHECK (wip_limit IS NULL OR wip_limit > 0)
);

CREATE INDEX idx_columns_tenant_board_pos
    ON board_columns (tenant_id, board_id, position);


-- Bảng thẻ công việc
CREATE TABLE cards (
    id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id     UUID NOT NULL
                  REFERENCES tenants(id) ON DELETE CASCADE,
    column_id     UUID NOT NULL
                  REFERENCES board_columns(id) ON DELETE CASCADE,
    assignee_id   UUID NULL
                  REFERENCES tenant_users(id) ON DELETE SET NULL,
    title         VARCHAR(200) NOT NULL,
    description   TEXT,
    position      INT NOT NULL DEFAULT 0,
    version       INT NOT NULL DEFAULT 1,
    custom_fields JSONB NOT NULL DEFAULT '{}'::jsonb,
    created_at    TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_cards_tenant_col_pos
    ON cards (tenant_id, column_id, position);

CREATE INDEX idx_cards_custom_fields_gin
    ON cards USING GIN (custom_fields jsonb_path_ops);
