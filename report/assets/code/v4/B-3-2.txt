// Thứ tự đọc: L1 Caffeine (Window-TinyLFU ~0.12ms) -> L2 Redis (~1.5ms) -> PostgreSQL DB Fallback
public void invalidateTenantConfig(UUID tenantId) {
    l1ConfigCache.invalidate(tenantId);                                             // 1. Xóa L1 cục bộ
    redisTemplate.delete("tenant:" + tenantId + ":config");                        // 2. Xóa L2 cấu hình
    redisTemplate.convertAndSend("tenant:config:invalidate", tenantId.toString()); // 3. Bắn Pub/Sub đa nút
}

public void evictBoardCache(UUID tenantId, UUID boardId) {
    redisTemplate.delete("tenant:" + tenantId + ":board:" + boardId); // Xóa đệm bảng theo khóa cô lập K2
}
