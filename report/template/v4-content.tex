  \renewcommand{\thefigure}{#1}%
  \begin{figure}[htbp]
    \centering
    \includegraphics[width=\linewidth,height=\VfourImageMaxHeight,keepaspectratio]{#2}%

  \end{figure}
  \endgroup}
